cmake_minimum_required(VERSION 3.16)
project(robot_execution)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(robot_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
get_filename_component(WORKSPACE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CARM_SDK_ROOT "${WORKSPACE_ROOT}/vendor/arm_control_sdk" CACHE PATH "CARM SDK root")
set(JSONCPP_COMPAT_ROOT "${WORKSPACE_ROOT}/vendor/jsoncpp_compat/usr/lib/x86_64-linux-gnu" CACHE PATH "JsonCpp ABI 1 library")
find_package(arm_control_sdk REQUIRED CONFIG PATHS "${CARM_SDK_ROOT}/lib/cmake/arm_control_sdk" NO_DEFAULT_PATH)
if(NOT EXISTS "${JSONCPP_COMPAT_ROOT}/libjsoncpp.so.1")
 message(FATAL_ERROR "缺少兼容的 libjsoncpp.so.1；请检查硬件与 SDK 配置文档。")
endif()
add_executable(execution_node src/execution_node.cpp)
target_include_directories(execution_node PRIVATE include ${arm_control_sdk_INCLUDE_DIRS})
ament_target_dependencies(execution_node rclcpp rclcpp_action robot_interfaces sensor_msgs std_srvs)
target_link_libraries(execution_node ${arm_control_sdk_LIBRARIES} "${JSONCPP_COMPAT_ROOT}/libjsoncpp.so.1")
# SDK dependencies themselves lack RPATH. DT_RPATH resolves their transitive libraries.
target_link_options(execution_node PRIVATE "-Wl,--disable-new-dtags" "-Wl,-rpath-link,${CARM_SDK_ROOT}/lib" "-Wl,-rpath-link,${CARM_SDK_ROOT}/poco/lib" "-Wl,-rpath-link,${JSONCPP_COMPAT_ROOT}")
set_target_properties(execution_node PROPERTIES BUILD_RPATH "${CARM_SDK_ROOT}/lib;${CARM_SDK_ROOT}/poco/lib;${JSONCPP_COMPAT_ROOT}" INSTALL_RPATH "${CARM_SDK_ROOT}/lib;${CARM_SDK_ROOT}/poco/lib;${JSONCPP_COMPAT_ROOT}")
install(TARGETS execution_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_settling test/test_settling.cpp)
 target_include_directories(test_settling PRIVATE include)
 ament_add_gtest(test_command_errors test/test_command_errors.cpp)
 target_include_directories(test_command_errors PRIVATE include)
 ament_add_gtest(test_validation test/test_validation.cpp)
 target_include_directories(test_validation PRIVATE include)
 ament_add_gtest(test_gripper_progress test/test_gripper_progress.cpp)
 target_include_directories(test_gripper_progress PRIVATE include)
endif()
ament_package()
